cmake_minimum_required(VERSION 3.25)
project(SqlitePoc LANGUAGES CXX)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Sql)
if(ANDROID)
    add_library(SqlitePoc SHARED src/android/SqlitePoc.cpp src/shared/SqliteDemo.cpp)
    target_link_libraries(SqlitePoc PRIVATE Qt6::Core Qt6::Sql)
elseif(IOS)
    add_library(SqlitePoc STATIC src/ios/SqlitePocBridge.cpp src/shared/SqliteDemo.cpp)
    target_link_libraries(SqlitePoc PRIVATE Qt6::Core Qt6::Sql Qt6::QSQLiteDriverPlugin)
endif()
